\begin{center}
  {\Large\bfseries Kevin Da Nendra dan \textit{Through Pass} Penentu Kemenangan}
\end{center}
\vspace{0.3cm}

\begin{center}
  \includegraphics[width=0.42\textwidth]{kevin-da-nendra.jpg}\\[3pt]
  {\small\textit{Kevin Da Nendra, gelandang serang Manchester City yang dijuluki \emph{``sang arsitek''}.}}
\end{center}
\vspace{0.3cm}

Menit ke-$88$ di Etihad Stadium, papan skor menyala \textbf{2--2}. Di tengah
gemuruh puluhan ribu penonton, \textbf{Kevin Da Nendra}, gelandang serang
Manchester City yang dijuluki \emph{``sang arsitek''} karena visinya membaca
ruang seperti membaca peta, menerima bola di tengah lapangan. Sekejap matanya
menangkap seorang rekan yang melakukan \emph{diagonal run}, menusuk celah sempit
di antara dua bek lawan. Celah itu hanya terbuka satu tarikan napas: salah sudut
sedikit bola dipotong, kurang tenaga sedikit bola keburu direbut kiper. Kevin
harus memutuskan arah, sudut, dan kecepatan umpan \textbf{seketika itu juga}.

\smallskip
Dari pelacakan \emph{tracking camera} (satuan meter), Kevin berada di titik
$K(12,\,8)$ dan rekan penerima di $P(34,\,25)$, sementara dua bek lawan menjaga
celah pada $B_1(21,\,15)$ dan $B_2(27,\,20)$. Kevin melepas umpan dengan
\textbf{kelajuan awal $28$ m/s} yang meluruh akibat gesekan udara menurut
$v(t)=28\,(0{,}94)^{t}$ (dengan $t$ dalam detik), dan bola meninggalkan kakinya
pada \textbf{sudut elevasi $32^\circ$} terhadap permukaan lapangan. Pelatih
menegaskan: sebuah \textit{through pass} disebut \textbf{efektif} apabila
kecepatan bola saat diterima \emph{lebih dari} $20$ m/s \emph{dan} arah umpan
berada pada rentang \textbf{$30^\circ$ sampai $45^\circ$} terhadap garis
horizontal lapangan.

\smallskip
Ia menarik kaki ke belakang\dots\ dan melepas sebuah \textit{through pass} yang
akan dibicarakan berhari-hari. Apakah secara \textbf{matematis} umpan itu memang
sempurna, atau Kevin baru saja mengambil risiko paling nekat dalam kariernya?
Buktikan melalui soal-soal berikut.

% ---------------------------- NOMOR 1 ----------------------------
\begin{soal}{1 \normalfont\itshape(Logaritma \& Eksponen)}
Tinjau model peluruhan kelajuan bola $v(t)=28\,(0{,}94)^{t}$ (m/s). Menurut
pelatih, bola masih ``bertenaga'' selama kelajuannya di atas $20$ m/s. Tentukan
Benar/Salah tiap pernyataan berikut.
\begin{bstabel}
  \bs{(1) Kelajuan awal bola (saat $t=0$) adalah $28$ m/s.}
  \bs{(2) Setelah $3$ detik, kelajuan bola $v(3)\approx 23{,}3$ m/s.}
  \bs{(3) Kelajuan bola turun tepat menjadi $20$ m/s pada sekitar $t\approx 5{,}4$ detik.}
  \bs{(4) Pada $t=10$ detik, kelajuan bola masih di atas $20$ m/s.}
\end{bstabel}
\end{soal}

\begin{soal}{2 \normalfont\itshape(Trigonometri: penguraian vektor kecepatan)}
Bola diangkat dengan kelajuan awal $28$ m/s pada sudut elevasi $32^\circ$.
Tentukan Benar/Salah tiap pernyataan berikut.

membentuk sudut $\theta$ terhadap horizontal dengan
$\tan\theta=\dfrac{25-8}{34-12}=\dfrac{17}{22}$. Tentukan Benar/Salah tiap
pernyataan berikut.

\end{center}

\begin{bstabel}
  \bs{(1) $\tan\theta=\dfrac{17}{22}$ sehingga $\theta\approx 37{,}7^\circ$.}
  \bs{(2) Arah umpan berada dalam rentang efektif $30^\circ$--$45^\circ$.}
  \bs{(3) Besar sudut $\theta$ tepat sama dengan sudut elevasi $32^\circ$.}

\end{soal}

% ---------------------------- NOMOR 4 ----------------------------
\begin{soal}{4 \normalfont\itshape(Vektor)}
Lintasan umpan dinyatakan sebagai vektor $\vec{KP}=P-K$. Tentukan Benar/Salah

\begin{bstabel}
  \bs{(1) $\vec{KP}=(22,\,17)$.}
  \bs{(2) Panjang umpan $|\vec{KP}|=\sqrt{773}\approx 27{,}8$ m.}
  \bs{(3) Vektor satuan searah umpan $\approx (0{,}79,\ 0{,}61)$.}
  \bs{(4) Panjang umpan sama dengan $22+17=39$ m.}
\end{bstabel}
\end{soal}

% ---------------------------- NOMOR 5 ----------------------------
\begin{soal}{5 \normalfont\itshape(Kolaborasi: kelayakan umpan)}
Dari rekaman video, bola diterima $1{,}2$ detik setelah ditendang. Dengan
menggabungkan hasil Nomor 1--4 (dan diketahui jarak garis lintasan ke $B_2$
adalah $\tfrac{9}{\sqrt{773}}\approx 0{,}32$ m serta ke $B_1$ adalah

\begin{bstabel}
  \bs{(1) Kecepatan saat diterima $v(1{,}2)\approx 26{,}0$ m/s, sehingga $>20$ m/s.}
  \bs{(2) Karena arah $37{,}7^\circ$ dan kecepatan terima $>20$ m/s, umpan memenuhi kedua kriteria efektif.}
  \bs{(3) Bola melintas $\approx 0{,}32$ m dari bek $B_2$.}
  \bs{(4) Karena jaraknya hanya $\approx 0{,}036$ m, bola melintas dengan aman jauh dari bek $B_1$.}
\end{bstabel}
\end{soal}

  \item Tuliskan kelajuan bola pada $t=0,1,2,3$ detik. Tunjukkan bahwa nilai-nilai
        itu membentuk barisan geometri dan tentukan rasionya.
  \item Dengan logaritma, tentukan pada detik ke berapa kelajuan bola tepat
        menjadi $20$ m/s.
  \item Periksa apakah pada $t=5$ detik bola masih ``bertenaga'' ($>20$ m/s)
        menggunakan $0{,}94^{5}\approx0{,}734$.
  \item Bila rekan baru menerima bola pada $t=1{,}2$ detik, tentukan kecepatan
        bola saat diterima dan simpulkan apakah masih memenuhi syarat kecepatan
        efektif ($>20$ m/s). Gunakan $0{,}94^{1{,}2}\approx0{,}928$.
\end{subsoal}
\end{soal}

\begin{soal}{12 \normalfont\bfseries[BONUS] \normalfont\itshape(Operasi dengan Berbagai Notasi)}
Kerjakan keempat bagian berikut. Definisi setiap lambang yang digunakan diberikan

\begin{subsoal}
  \item \textbf{(Produk $\prod$.)} Tiga kejadian saling bebas dengan
        peluang $p_1=0{,}75$, $p_2=0{,}80$, $p_3=0{,}50$. Lambang $\prod$ berarti
        ``kalikan semua''. Hitung $P(\text{gol})=\displaystyle\prod_{i=1}^{3}p_i$.
        \begin{center}
        \begin{tikzpicture}[>=Latex,font=\scriptsize]
          \node (a) at (0,0){mulai};

          \draw[->] (b)--(c) node[midway,above]{$0{,}80$};

        \end{tikzpicture}

  \item \textbf{(Determinan.)} Titik perebutan bola memenuhi
        $\begin{cases}x+y=44\\ 3x-2y=37\end{cases}$. Selesaikan dengan
        \emph{aturan Cramer} (cukup pola $ad-bc$ pada tiap determinan):
        \[
        D=\begin{vmatrix}1&1\\ 3&-2\end{vmatrix},\quad
        D_x=\begin{vmatrix}44&1\\ 37&-2\end{vmatrix},\quad
        D_y=\begin{vmatrix}1&44\\ 3&37\end{vmatrix},\quad
        x=\frac{D_x}{D},\ \ y=\frac{D_y}{D}.

        Hitung $x$ dan $y$. \Big(Ingat $\begin{vmatrix}a&b\\ c&d\end{vmatrix}=ad-bc$.\Big)
  \item \textbf{(Notasi himpunan.)} Tuliskan semua anggota
        \[
        S=\{\,x\in\mathbb{Z}\ \mid\ 30\le x\le 45\ \text{dan}\ x\equiv 2\ (\mathrm{mod}\ 5)\,\}.

        (``$x\equiv 2\ (\mathrm{mod}\ 5)$'' artinya sisa bagi $x$ oleh $5$ sama dengan $2$.)
